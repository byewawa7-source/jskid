# jStore — Community Mod Store

## 1. Overview
jStore is a community-driven mod marketplace hosted on GitHub. Users can browse, install, and publish mods directly from within JanitorAI.

## 4. Store Client (src/store/store.js)

### Core Functions
```javascript
class StoreClient {
  constructor() {
    this.storeRepo = "jskid/jskid-store";
    this.indexUrl = `https://raw.githubusercontent.com/jskid/jskid-store/main/index.json`;
    this.cache = null;
    this.cacheTime = null;
    this.cacheDuration = 5 * 60 * 1000; // 5 minutes
  }
  
  // Fetch the mod index
  async fetchIndex(force = false) {
    if (!force && this.cache && Date.now() - this.cacheTime < this.cacheDuration) {
      return this.cache;
    }
    const response = await fetch(this.indexUrl);
    this.cache = await response.json();
    this.cacheTime = Date.now();
    return this.cache;
  }
  
  // Get mod details
  async getMod(modId) {
    const index = await this.fetchIndex();
    return index.mods.find(m => m.id === modId);
  }
  
  // Get mod source code
  async getModSource(modId) {
    const mod = await this.getMod(modId);
    const url = `https://raw.githubusercontent.com/jskid/jskid-store/main/${mod.path}main.jsk`;
    const response = await fetch(url);
    return await response.text();
  }
  
  // Get mod manifest
  async getModManifest(modId) {
    const mod = await this.getMod(modId);
    const url = `https://raw.githubusercontent.com/jskid/jskid-store/main/${mod.path}manifest.json`;
    const response = await fetch(url);
    return await response.json();
  }
  
  // Search mods
  searchMods(query, filters = {}) {
    let results = this.cache.mods;
    if (query) {
      const q = query.toLowerCase();
      results = results.filter(m => 
        m.name.toLowerCase().includes(q) ||
        m.description.toLowerCase().includes(q) ||
        m.tags.some(t => t.includes(q)) ||
        m.author.toLowerCase().includes(q)
      );
    }
    if (filters.type) {
      results = results.filter(m => m.type === filters.type);
    }
    if (filters.category) {
      results = results.filter(m => m.tags.includes(filters.category));
    }
    if (filters.sort === "downloads") {
      results.sort((a, b) => b.downloads - a.downloads);
    } else if (filters.sort === "rating") {
      results.sort((a, b) => b.rating - a.rating);
    } else if (filters.sort === "newest") {
      results.sort((a, b) => new Date(b.updatedAt) - new Date(a.updatedAt));
    }
    return results;
  }
}
```

## 5. Installer (src/store/installer.js)
```javascript
class ModInstaller {
  constructor(modManager) {
    this.modManager = modManager;
  }
  
  async install(modId) {
    // 1. Fetch manifest
    const manifest = await storeClient.getModManifest(modId);
    
    // 2. Check dependencies
    if (manifest.dependencies) {
      for (const dep of manifest.dependencies) {
        if (!this.modManager.isInstalled(dep.id)) {
          // Prompt user to install dependency first
          await this.install(dep.id);
        }
      }
    }
    
    // 3. Check conflicts
    if (manifest.conflicts) {
      for (const conflict of manifest.conflicts) {
        if (this.modManager.isInstalled(conflict)) {
          throw new Error(`Mod conflicts with installed mod: ${conflict}`);
        }
      }
    }
    
    // 4. Show permission prompt
    const granted = await this.showPermissionPrompt(manifest);
    if (!granted) return;
    
    // 5. Download source
    const source = await storeClient.getModSource(modId);
    
    // 6. Install
    await this.modManager.install(manifest, source);
    
    // 7. Track download
    this.trackDownload(modId);
  }
  
  async showPermissionPrompt(manifest) {
    // Show modal with permission list
    // Return true/false based on user choice
    return new Promise((resolve) => {
      const modal = new PermissionModal(manifest, resolve);
      modal.show();
    });
  }
  
  async uninstall(modId) {
    await this.modManager.uninstall(modId);
  }
  
  async checkForUpdates() {
    const installed = this.modManager.getInstalledMods();
    const updates = [];
    for (const mod of installed) {
      const storeMod = await storeClient.getMod(mod.id);
      if (storeMod && this.compareVersions(storeMod.version, mod.version) > 0) {
        updates.push({ id: mod.id, current: mod.version, latest: storeMod.version });
      }
    }
    return updates;
  }
  
  compareVersions(a, b) {
    const pa = a.split('.').map(Number);
    const pb = b.split('.').map(Number);
    for (let i = 0; i < 3; i++) {
      if (pa[i] > pb[i]) return 1;
      if (pa[i] < pb[i]) return -1;
    }
    return 0;
  }
}
```

## 8. Auto-Update System
```javascript
class ModUpdater {
  constructor(installer) {
    this.installer = installer;
    this.checkInterval = 24 * 60 * 60 * 1000; // Daily
  }
  
  start() {
    // Check for updates on startup
    this.check();
    // Periodic check
    setInterval(() => this.check(), this.checkInterval);
  }
  
  async check() {
    const updates = await this.installer.checkForUpdates();
    if (updates.length > 0) {
      this.showUpdateNotification(updates);
    }
  }
  
  showUpdateNotification(updates) {
    // Show notification with update count
    // User can click to update all or individual
  }
  
  async updateAll() {
    const updates = await this.installer.checkForUpdates();
    for (const update of updates) {
      await this.installer.install(update.id);
    }
  }
}
```

## 9. Store CI/CD (GitHub Actions)
```yaml
# .github/workflows/build.yml
name: Build Store Index
on:
  push:
    branches: [main]
    paths:
      - 'mods/**'
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Validate mods
        run: node scripts/validate-mod.js
      - name: Build index
        run: node scripts/build-index.js
      - name: Commit index
        run: |
          git config user.name "jStore Bot"
          git config user.email "bot@jskid.dev"
          git add index.json
          git commit -m "Update store index" || exit 0
          git push
```

## 10. Rating & Review System
- Users can rate mods (1-5 stars)
- Ratings stored in a separate GitHub-backed system
- Optional: review comments
- Rating shown in store browser and detail view
- Anti-abuse: one rating per user per mod version
